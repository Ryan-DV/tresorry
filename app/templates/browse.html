Command line access

HTTP Basic auth works with any username and the passphrase. Use -k because the certificate is self-signed.

# list this folder
curl -k -u :'PASSPHRASE' https://{{ host }}/browse/{{ rel }}

# download
curl -k -u :'PASSPHRASE' -OJ https://{{ host }}/dl/{{ rel ~ '/' if rel }}FILE

# upload into this folder
curl -k -u :'PASSPHRASE' -F file=@./loot.zip https://{{ host }}/upload/{{ rel }}

# wget / PowerShell
wget --no-check-certificate --auth-no-challenge --user=x --password='PASSPHRASE' https://{{ host }}/dl/{{ rel ~ '/' if rel }}FILE
iwr -SkipCertificateCheck -Credential (Get-Credential x) https://{{ host }}/dl/{{ rel ~ '/' if rel }}FILE -OutFile FILE
